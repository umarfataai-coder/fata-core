# Amfani da tushen Python mai sauri tare da tallafin PyTorch da CUDA
FROM python:3.10-slim

# Sanya kayan aikin tsari
RUN apt-get update && apt-get install -y \
    build-essential \
    curl \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /fata_app

# Kwafi da sanya dukkan dependencies
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Kwafi lambobin Fata AI
COPY . .

# Buɗe hanyar sadarwa (Port) ta FastAPI
EXPOSE 8000

# Fara gudanar da Uvicorn Server
CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "4"]
